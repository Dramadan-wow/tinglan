cmake_minimum_required(VERSION 3.16)
project(InstantMessengerHsx LANGUAGES CXX)
file(STRINGS "${CMAKE_SOURCE_DIR}/VERSION" IM_APP_VERSION LIMIT_COUNT 1)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/VERSION")
add_compile_definitions(IM_APP_VERSION="${IM_APP_VERSION}")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
file(GLOB CORE_SOURCES CONFIGURE_DEPENDS "src/*Hsx.cpp")
add_library(im_core STATIC ${CORE_SOURCES})
target_include_directories(im_core PUBLIC src)
if(MSVC)
    target_compile_options(im_core PUBLIC /utf-8 /W4 /permissive-)
else()
    target_compile_options(im_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(WIN32)
    target_link_libraries(im_core PUBLIC ws2_32 bcrypt)
endif()
add_executable(im_system src/main.cpp)
target_link_libraries(im_system PRIVATE im_core)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE im_core)
add_test(NAME scenarios COMMAND core_tests)
if(WIN32)
    add_executable(admin_tests tests/admin_tests.cpp)
    target_link_libraries(admin_tests PRIVATE im_core)
    add_test(NAME admin_scenarios COMMAND admin_tests)
endif()

option(IM_BUILD_GUI "Build the Qt Widgets desktop interface" OFF)
option(IM_BUILD_SERVER "Build the independent central chat server" OFF)
if(IM_BUILD_GUI OR IM_BUILD_SERVER)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network)
    target_compile_definitions(im_core PRIVATE IM_QT_CRYPTO)
    target_link_libraries(im_core PUBLIC Qt6::Core Qt6::Network)
    add_library(im_workspace STATIC workspace/WorkspaceCodecHsx.cpp workspace/AppearanceHsx.cpp)
    target_include_directories(im_workspace PUBLIC workspace)
    target_link_libraries(im_workspace PUBLIC im_core Qt6::Core)
    add_executable(workspace_data_tool tests/workspace_data_tool.cpp)
    target_link_libraries(workspace_data_tool PRIVATE im_workspace)
    add_executable(workspace_privacy_checks tests/workspace_privacy_checks.cpp)
    target_link_libraries(workspace_privacy_checks PRIVATE im_workspace)
    add_test(NAME workspace_privacy COMMAND workspace_privacy_checks)
    add_subdirectory(network)
    add_executable(update_client_probe tests/update_client_probe.cpp)
    target_link_libraries(update_client_probe PRIVATE im_network)
    add_executable(chat_client_checks tests/chat_client_checks.cpp)
    target_link_libraries(chat_client_checks PRIVATE im_network)
    add_test(NAME chat_client COMMAND chat_client_checks)
    add_executable(qr_code_checks tests/qr_code_checks.cpp)
    target_link_libraries(qr_code_checks PRIVATE im_network)
    add_test(NAME qr_code COMMAND qr_code_checks)
endif()
if(IM_BUILD_SERVER)
    add_subdirectory(server)
    add_executable(appearance_server_checks tests/appearance_server_checks.cpp)
    target_link_libraries(appearance_server_checks PRIVATE im_server_core)
    add_test(NAME appearance_server COMMAND appearance_server_checks)
    add_executable(session_server_checks tests/session_server_checks.cpp)
    target_link_libraries(session_server_checks PRIVATE im_server_core)
    add_test(NAME session_server COMMAND session_server_checks)
endif()
if(IM_BUILD_GUI)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network Multimedia)

        set(IM_PAYMENT_RESOURCE "${IM_ASSET_ROOT}/payment/wechat-receive.jpg")

        set(IM_PAYMENT_RESOURCE "${IM_ASSET_ROOT}/payment/placeholder.xpm")
    endif()

    set(IM_LOGIN_RESOURCE "${CMAKE_CURRENT_BINARY_DIR}/login.qrc")
    file(GLOB GUI_SOURCES CONFIGURE_DEPENDS "gui/*Hsx.cpp" "gui/*Hsx.h")
    add_library(im_desktop_ui STATIC ${GUI_SOURCES})
    set_target_properties(im_desktop_ui PROPERTIES AUTOMOC ON)
    target_include_directories(im_desktop_ui PUBLIC gui)
    target_link_libraries(im_desktop_ui PUBLIC im_core im_network im_workspace Qt6::Widgets Qt6::Network Qt6::Multimedia)
    set(CLIENT_GUI_SOURCES ${GUI_SOURCES})

    set_target_properties(im_client_ui PROPERTIES AUTOMOC ON)
    target_compile_definitions(im_client_ui PUBLIC IM_USER_CLIENT)
    target_include_directories(im_client_ui PUBLIC gui)
    target_link_libraries(im_client_ui PUBLIC im_core im_network im_workspace Qt6::Widgets Qt6::Network Qt6::Multimedia)
    add_executable(im_client WIN32 gui/client_main.cpp "${IM_LOGIN_RESOURCE}")
    set_target_properties(im_client PROPERTIES AUTORCC ON)
    target_link_libraries(im_client PRIVATE im_client_ui)
    add_executable(update_gui_checks tests/update_gui_checks.cpp "${IM_LOGIN_RESOURCE}")
    add_executable(file_client_probe tests/file_client_probe.cpp)
    target_link_libraries(file_client_probe PRIVATE im_client_ui)
    set_target_properties(update_gui_checks PROPERTIES AUTORCC ON)
    target_link_libraries(update_gui_checks PRIVATE im_client_ui)
    add_executable(im_desktop WIN32 gui/main.cpp "${IM_LOGIN_RESOURCE}")
    set_target_properties(im_desktop PROPERTIES AUTORCC ON)
    target_link_libraries(im_desktop PRIVATE im_desktop_ui)
    add_executable(connection_code_checks tests/connection_code_checks.cpp "${IM_LOGIN_RESOURCE}")
    set_target_properties(connection_code_checks PROPERTIES AUTORCC ON)
    target_link_libraries(connection_code_checks PRIVATE im_client_ui)
    add_executable(client_gui_probe tests/client_gui_probe.cpp "${IM_LOGIN_RESOURCE}")

    add_executable(network_gui_probe tests/network_gui_probe.cpp "${IM_LOGIN_RESOURCE}")
    set_target_properties(network_gui_probe PROPERTIES AUTORCC ON)
    target_link_libraries(network_gui_probe PRIVATE im_desktop_ui)
    add_executable(workspace_gui_probe tests/workspace_gui_probe.cpp "${IM_LOGIN_RESOURCE}")
    set_target_properties(workspace_gui_probe PROPERTIES AUTORCC ON)
    target_link_libraries(workspace_gui_probe PRIVATE im_desktop_ui)
    add_executable(connection_gui_checks tests/connection_gui_checks.cpp "${IM_LOGIN_RESOURCE}")
    set_target_properties(connection_gui_checks PROPERTIES AUTORCC ON)
    target_link_libraries(connection_gui_checks PRIVATE im_desktop_ui)
    set(IM_DESKTOP_OUTPUT "${CMAKE_SOURCE_DIR}/bin" CACHE PATH "Desktop runtime output")
    set_target_properties(im_desktop im_client PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY_RELEASE "${IM_DESKTOP_OUTPUT}"
        RUNTIME_OUTPUT_DIRECTORY_DEBUG "${CMAKE_BINARY_DIR}/Debug")
