cmake_minimum_required(VERSION 3.21)
project(TinglanPortable LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
if(NOT DEFINED PAYLOAD OR NOT DEFINED PAYLOAD_HASH)
  message(FATAL_ERROR "PAYLOAD and PAYLOAD_HASH required")
endif()
file(TO_CMAKE_PATH "${PAYLOAD}" PAYLOAD_RESOURCE)
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/payload.rc" "#pragma code_page(65001)\n101 RCDATA \"${PAYLOAD_RESOURCE}\"\n")
add_executable(TinglanPortable WIN32 launcher.cpp "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
target_compile_options(TinglanPortable PRIVATE /utf-8)
target_compile_definitions(TinglanPortable PRIVATE PAYLOAD_VERSION=L"${PAYLOAD_HASH}")
target_link_libraries(TinglanPortable PRIVATE shell32)
