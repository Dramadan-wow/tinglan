cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(TinglanServer LANGUAGES CXX)
    set(CMAKE_CXX_STANDARD 17)
    set(CMAKE_CXX_STANDARD_REQUIRED ON)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Core Network Sql Gui)
    file(GLOB CORE_SOURCES "../src/*Hsx.cpp")
    add_library(im_core STATIC ${CORE_SOURCES})
    target_include_directories(im_core PUBLIC ../src)
    target_compile_definitions(im_core PRIVATE IM_QT_CRYPTO)
    target_link_libraries(im_core PUBLIC Qt6::Core Qt6::Network)
    if(MSVC)
        target_compile_options(im_core PUBLIC /utf-8)
        target_link_libraries(im_core PUBLIC ws2_32 bcrypt)
    endif()
    add_library(im_workspace STATIC ../workspace/WorkspaceCodecHsx.cpp ../workspace/AppearanceHsx.cpp)
    target_include_directories(im_workspace PUBLIC ../workspace)
    target_link_libraries(im_workspace PUBLIC im_core Qt6::Core)
    add_subdirectory(../network ${CMAKE_BINARY_DIR}/network)
endif()
find_package(Qt6 6.8 REQUIRED COMPONENTS Sql Gui)
add_library(im_server_core STATIC AuthStoreHsx.cpp AuthStoreHsx.h ChatServerHsx.cpp ChatServerHsx.h WorkspaceBackendHsx.cpp WorkspaceBackendHsx.h SocialStoreHsx.cpp SocialStoreHsx.h UpdateStoreHsx.cpp)
set_target_properties(im_server_core PROPERTIES AUTOMOC ON)
target_include_directories(im_server_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(im_server_core PUBLIC im_network im_workspace Qt6::Sql Qt6::Gui)
add_executable(im_server main.cpp)
target_link_libraries(im_server PRIVATE im_server_core)
